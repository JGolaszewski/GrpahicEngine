200template <typename T> inline LogArgument MakeArgument(T &&input) noexcept
201{
202 using ValueType = std::remove_cvref_t<T>;
203 LogArgument result;
204 if constexpr (std::is_same_v<ValueType, bool>)
205 {
206 result.type = ArgumentType::Boolean;
207 result.value.booleanValue = input;
208 }
209 else if constexpr (std::is_enum_v<ValueType>)
210 {
211 using Underlying = std::underlying_type_t<ValueType>;
212 return MakeArgument(static_cast<Underlying>(input));
213 }
214 else if constexpr (std::is_integral_v<ValueType> && std::is_signed_v<ValueType>)
215 {
216 result.type = ArgumentType::Signed;
217 result.value.signedValue = static_cast<int64_t>(input);
218 }
219 else if constexpr (std::is_integral_v<ValueType>)
220 {
221 result.type = ArgumentType::Unsigned;
222 result.value.unsignedValue = static_cast<uint64_t>(input);
223 }
224 else if constexpr (std::is_floating_point_v<ValueType>)
225 {
226 result.type = ArgumentType::Floating;
227 result.value.floatingValue = static_cast<double>(input);
228 }
229 else if constexpr (std::convertible_to<T, std::string_view>)
230 {
231 result.type = ArgumentType::Text;
232 const std::string_view view(input);
233 const size_t count = (std::min)(view.size(), MaxLogArgumentText - 1);
234 if (count > 0)
235 std::copy_n(view.data(), count, result.value.text);
236 result.value.text[count] = '\0';
237 }
238 else
239 {
240 static_assert(sizeof(T) == 0, "Log arguments must be primitive, enum, or string-like values");
241 }
242 return result;
243}
