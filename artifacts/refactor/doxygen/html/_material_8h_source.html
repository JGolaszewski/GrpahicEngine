147inline MaterialGpu BuildMaterialGpu(const MaterialProperties &properties) noexcept
148{
149 const float roughness = ClampFinite(properties.Roughness, 0.5f, 0.0f, 1.0f);
150 const float ior = ClampFinite(properties.Ior, 1.5f, MaterialMinIor, MaterialMaxIor);
151 const auto validModel = [](BrdfModel model) {
152 return static_cast<uint32_t>(model) <= static_cast<uint32_t>(BrdfModel::GTR) ? model
153 : BrdfModel::Legacy;
154 };
155 const auto validWorkflow = [](MaterialWorkflow workflow) {
156 return static_cast<uint32_t>(workflow) <= static_cast<uint32_t>(MaterialWorkflow::SpecularGlossiness)
157 ? workflow
158 : MaterialWorkflow::MetallicRoughness;
159 };
160 const auto validSpecularMode = [](DielectricSpecularMode mode) {
161 return static_cast<uint32_t>(mode) <= static_cast<uint32_t>(DielectricSpecularMode::ExplicitF0)
162 ? mode
163 : DielectricSpecularMode::IndexOfRefraction;
164 };
165 const auto validAlphaMode = [](MaterialAlphaMode mode) {
166 return static_cast<uint32_t>(mode) <= static_cast<uint32_t>(MaterialAlphaMode::Blend)
167 ? mode
168 : MaterialAlphaMode::Opaque;
169 };

171 MaterialGpu gpu{
172 {ClampFinite(properties.BaseColor.x, 1.0f, 0.0f, 1.0f),
173 ClampFinite(properties.BaseColor.y, 1.0f, 0.0f, 1.0f),
174 ClampFinite(properties.BaseColor.z, 1.0f, 0.0f, 1.0f),
175 ClampFinite(properties.BaseColor.w, 1.0f, 0.0f, 1.0f)},
176 {FiniteNonNegative(properties.Emissive.x), FiniteNonNegative(properties.Emissive.y),
177 FiniteNonNegative(properties.Emissive.z), ClampFinite(properties.Metallic, 0.0f, 0.0f, 1.0f)},
178 {roughness, (std::max)(roughness * roughness, MaterialMinAlpha),
179 ClampFinite(properties.GtrGamma, 2.0f, MaterialMinGtrGamma, MaterialMaxGtrGamma), ior},
180 {ClampFinite(properties.SpecularF0.x, 0.04f, 0.0f, 1.0f),
181 ClampFinite(properties.SpecularF0.y, 0.04f, 0.0f, 1.0f),
182 ClampFinite(properties.SpecularF0.z, 0.04f, 0.0f, 1.0f), 0.0f},
183 {MaterialGpuAbiVersion, static_cast<uint32_t>(validModel(properties.Model)),
184 static_cast<uint32_t>(validSpecularMode(properties.SpecularMode)) |
185 (validWorkflow(properties.Workflow) == MaterialWorkflow::SpecularGlossiness ? 0x100u : 0u),
186 static_cast<uint32_t>(validAlphaMode(properties.AlphaMode)) |
187 (properties.DoubleSided ? 0x100u : 0u)},
188 {ClampFinite(properties.AlphaCutoff, 0.5f, 0.0f, 1.0f), 0.0f, 0.0f, 0.0f}};
189 gpu.TextureCoordinates0 = {
190 properties.BaseColorTexCoord < MaxTextureCoordinateSets ? properties.BaseColorTexCoord : 0,
191 properties.NormalTexCoord < MaxTextureCoordinateSets ? properties.NormalTexCoord : 0,
192 properties.MetallicRoughnessTexCoord < MaxTextureCoordinateSets ? properties.MetallicRoughnessTexCoord
193 : 0,
194 properties.OcclusionTexCoord < MaxTextureCoordinateSets ? properties.OcclusionTexCoord : 0};
195 gpu.TextureCoordinates1 = {
196 properties.EmissiveTexCoord < MaxTextureCoordinateSets ? properties.EmissiveTexCoord : 0, 0, 0, 0};
197 gpu.TextureParams = {ClampFinite(properties.NormalScale, 1.0f, 0.0f, 1.0e6f),
198 ClampFinite(properties.OcclusionStrength, 1.0f, 0.0f, 1.0f), 0.0f, 0.0f};
199 for (size_t i = 0; i < 5; ++i)
200 {
201 const auto &transform = properties.TextureTransforms[i];
202 const float rotation = std::isfinite(transform.Rotation) ? transform.Rotation : 0.0f;
203 const float cosine = std::cos(rotation);
204 const float sine = std::sin(rotation);
205 const float scaleX = std::isfinite(transform.Scale.x) ? transform.Scale.x : 1.0f;
206 const float scaleY = std::isfinite(transform.Scale.y) ? transform.Scale.y : 1.0f;
207 gpu.TextureTransformRows[i] = {cosine * scaleX, -sine * scaleY, sine * scaleX, cosine * scaleY};
208 gpu.TextureTransformOffsets[i] = {std::isfinite(transform.Offset.x) ? transform.Offset.x : 0.0f,
209 std::isfinite(transform.Offset.y) ? transform.Offset.y : 0.0f, 0.0f,
210 0.0f};
211 }
212 return gpu;
